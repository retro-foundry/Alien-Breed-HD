cmake_minimum_required(VERSION 3.16)
project(ab3d_render_regressions LANGUAGES C)
set(CMAKE_C_STANDARD 11)
get_filename_component(AB3D_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${AB3D_ROOT}" engine EXCLUDE_FROM_ALL)
include("${AB3D_ROOT}/cmake/game_sources.cmake")
list(REMOVE_ITEM SOURCES src/main.c src/renderer.c src/display.c)
list(TRANSFORM SOURCES PREPEND "${AB3D_ROOT}/")
add_executable(weapon_water_cpu ../feedback_weapon_water.c ${SOURCES} "${AB3D_ROOT}/src/display.c")
add_executable(weapon_water_display ../weapon_water_display.c ../feedback_weapon_water.c ${SOURCES})
add_executable(renderer_tags ../feedback_tags.c ${SOURCES} "${AB3D_ROOT}/src/display.c")
target_compile_definitions(renderer_tags PRIVATE AB3D_TAGS_EXTERNAL_STATE)
add_executable(desktop_scenes EXCLUDE_FROM_ALL ../desktop_scenes.c ${SOURCES}
  "${AB3D_ROOT}/src/renderer.c" "${AB3D_ROOT}/src/display.c")
target_compile_definitions(weapon_water_display PRIVATE AB3D_WEAPON_WATER_SCENE_LIB)
foreach(target weapon_water_cpu weapon_water_display renderer_tags desktop_scenes)
  target_include_directories(${target} PRIVATE "${AB3D_ROOT}/src" "${AB3D_ROOT}/third_party")
  target_compile_definitions(${target} PRIVATE SDL_MAIN_HANDLED AB3D_ENABLE_FLOOR_COL_FAST=1 AB3D_WEAPON_WATER_EXTERNAL_STATE)
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
  target_link_libraries(${target} PRIVATE SDL2-static)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${AB3D_TEST_DATA}" "$<TARGET_FILE_DIR:${target}>/data"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${AB3D_ROOT}/fonts" "$<TARGET_FILE_DIR:${target}>/fonts")
endforeach()
